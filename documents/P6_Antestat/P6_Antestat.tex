-- Funktion zur zufälligen Auswahl von Fragen
function selectRandomQuestions(n)
    math.randomseed(os.time())
    local selected = {}
    local indices = {}
    
    -- Erstelle Array mit allen Indizes
    for i = 1, #questions do
        indices[i] = i
    end
    
    -- Wähle n zufällige Fragen aus
    for i = 1, math.min(n, #questions) do
        local randomIndex = math.random(1, #indices)
        table.insert(selected, questions[indices[randomIndex]])
        table.remove(indices, randomIndex)
    end
    
    return selected
end

-- Hier wird die Anzahl der Fragen festgelegt, die ausgewählt werden sollen
selectedQuestions = selectRandomQuestions(11)

-- Funktionen für LaTeX-Output
function printQuestions()
    for i, q in ipairs(selectedQuestions) do
        tex.print("\\question{" .. i .. "}{" .. q.question .. "}")
    end
end

function printAnswers()
    for i, q in ipairs(selectedQuestions) do
        tex.print("\\answer{" .. i .. "}{" .. q.answer .. "}")
    end
end
