-- Funktion zur zufälligen Auswahl von Fragen
function selectRandomQuestionsA(n)
    math.randomseed(os.time())
    local selected = {}
    local indices = {}
    
    -- Erstelle Array mit allen Indizes
    for i = 1, #questionsA do
        indices[i] = i
    end
    
    -- Wähle n zufällige Fragen aus
    for i = 1, math.min(n, #questionsA) do
        local randomIndex = math.random(1, #indices)
        table.insert(selected, questionsA[indices[randomIndex]])
        table.remove(indices, randomIndex)
    end
    
    return selected
end
